A102469 人通过 · 位运算 · 你 AC 了（8 分钟）

题意：给正整数 n（n ≤ 10⁶），判断它是不是 2 的整数次幂。

思路：2 的幂的二进制里只有 1 个 1。所以 n & (n-1) 会把最低位的那个 1 消掉——结果是 0 就是 2 的幂。n=1 也满足（1 & 0 = 0），正好题目认为 1 是 2⁰。

A.c
#include <stdio.h>
int main(void) {
    int n;
    scanf("%d", &n);
    if (n > 0 && (n & (n - 1)) == 0) printf("YES\n");
    else                            printf("NO\n");
    return 0;
}

坑：这一题判题器是 Special Judge，输出 Yes/yes/YES 都行，别在这上面纠结。

B?! 回文文回 !?24 人通过 · 拆位 + 双指针 · 你 AC 了（22 分钟）

题意：给 n 个数（n ≤ 10⁵，aᵢ ≤ 10⁹），逐个判断它的二进制表示（去掉前导零）是不是回文。特别规定 0 的二进制是 0，是回文。

思路：把每一位拆到一个数组里（低位在前），然后首尾两个指针往中间夹。0 要单独判，因为它一位都不进循环。

B.c
#include <stdio.h>
int main(void) {
    int n, i;
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        unsigned int x; int b[40], k = 0, ok = 1, t;
        scanf("%u", &x);
        if (x == 0) { printf("Yes\n"); continue; }  /* 0 是回文 */
        while (x) { b[k++] = x & 1; x >>= 1; }        /* 低位在前 */
        for (t = 0; t < k / 2; t++)
            if (b[t] != b[k - 1 - t]) { ok = 0; break; }
        printf(ok ? "Yes\n" : "No\n");
    }
    return 0;
}

坑：0 一定要特判；aᵢ 用 unsigned int 或者直接用 int 都行（10⁹ < 2³¹）。

C阳光分配5 人通过 · 01 背包 + 方案还原 · 你没提交

题意：阳光总量 S，n 株植物（n ≤ 100，S ≤ 10⁵）。每株有消耗 cost 和防御值 val，每株只能买一次。求最大总防御值，并输出买了哪些（按消耗降序，同消耗按价值降序）。

思路：这就是 01 背包。但多了一层——要输出方案。一维 dp[j] 里只剩一个数字，回溯不出"哪几株被买了"，所以必须开二维。

为什么全场只有 5 人过 普通 01 背包很多人会背，但"记录选择"这一步要理解 dp[i][j] 的含义：只看前 i 株、阳光上限 j 时的最大防御值。有了第一维，才能从 (n, S) 往回走。
C.c
#include <stdio.h>

#define MAXN 105
#define MAXS 100005

int cost[MAXN], val[MAXN];
int dp[MAXN][MAXS];        /* 约 42MB，必须放全局 */
int pick[MAXN];

int main(void) {
    int n, S, i, j, k, cnt = 0;

    scanf("%d %d", &n, &S);
    for (i = 1; i <= n; i++) scanf("%d %d", &cost[i], &val[i]);

    for (i = 1; i <= n; i++) {
        for (j = 0; j <= S; j++) {
            dp[i][j] = dp[i-1][j];                        /* 不买第 i 株 */
            if (j >= cost[i] && dp[i-1][j-cost[i]] + val[i] > dp[i][j])
                dp[i][j] = dp[i-1][j-cost[i]] + val[i];   /* 买第 i 株 */
        }
    }

    /* 还原：从最后一格往回走 */
    j = S;
    for (i = n; i >= 1; i--) {
        if (dp[i][j] != dp[i-1][j]) {   /* 说明第 i 株被买了 */
            pick[cnt++] = i;
            j -= cost[i];
        }
    }

    printf("%d\n", dp[n][S]);

    /* 按消耗降序、同消耗按价值降序 */
    for (i = 0; i < cnt; i++)
        for (k = i + 1; k < cnt; k++)
            if (cost[pick[k]] > cost[pick[i]] ||
               (cost[pick[k]] == cost[pick[i]] && val[pick[k]] > val[pick[i]])) {
                int t = pick[i]; pick[i] = pick[k]; pick[k] = t;
            }

    for (i = 0; i < cnt; i++)
        printf("%d %d\n", cost[pick[i]], val[pick[i]]);

    return 0;
}

样例验证：4 100 / 50 60 / 30 40 / 20 25 / 40 50 → 输出 125，然后 50 60 / 30 40 / 20 25。

坑：① dp[i-1][...] 不是 dp[i][...]——这才是"每株只用一次"；② 还原时一定要先判 dp[i][j] != dp[i-1][j] 再减 cost；③ 数组 42MB，放 main 里会直接崩栈。

D我的世界均分食物7 人通过 · 前缀和 + 二分 · 你 4 次 WA 未过

题意：n 份食物，饱食度依次是 k, k+1, …, k+n-1。选一个切点 i，把前 i 份给左边、剩下的给右边，让两边总和的差的绝对值最小。n, k ≤ 10⁹。

思路：前 i 份的和 = i(2k+i-1)/2，记 F(i) = i(2k+i-1)（正好是两倍）。总和 2·S。目标就是让 F(i) 尽量接近 S。

F(i) 随 i 单调递增，所以二分找第一个 F(i) ≥ 总和的 i，答案一定在 i-1 和 i 里，两个都算一遍取最小。

D.c
#include <stdio.h>

int main(void) {
    long long n, k, total, lo, hi, mid, ans;

    scanf("%lld %lld", &n, &k);
    total = n * (2 * k + n - 1) / 2;          /* 全部饱食度之和 */

    lo = 1; hi = n - 1;                        /* 切点范围 1..n-1 */
    while (lo < hi) {
        mid = lo + (hi - lo) / 2;
        if (mid * (2 * k + mid - 1) >= total) hi = mid;
        else                                  lo = mid + 1;
    }

    ans = total;                               /* 先取一个上界 */
    for (mid = lo - 1; mid <= lo; mid++) {
        if (mid < 1 || mid > n - 1) continue;
        long long d = mid * (2 * k + mid - 1) - total;
        if (d < 0) d = -d;
        if (d < ans) ans = d;
    }

    printf("%lld\n", ans);
    return 0;
}

样例验证：7 2 → 5。

坑：① n·k 最大到 10¹⁸，必须 long long——你 4 次 WA 大概率死在这里；② 两倍关系别搞混：F(i) 是两倍前缀和，直接和 total 比，不要再多乘 2；③ 只比较 lo-1 和 lo 两个候选，不要从头枚举。

E群雄逐鹿（简单版）35 人通过 · 直接模拟 · 你 AC 了（63 分钟）

题意：n 座城市排成一排（n ≤ 5000）。m 次操作（m ≤ 5000），每次把区间 [u,v] 全染成颜色 c，后发生的覆盖先发生的。最后输出每座城市的颜色，没被染过的输出 0。

思路：数据小到 5000×5000 = 2.5×10⁷，直接暴力覆盖就行，不用任何技巧。这是全场性价比最高的题之一。

E.c
#include <stdio.h>

#define MAXN 5005
int own[MAXN];       /* 全局，自动清零，正好对应"没被占领输出 0" */

int main(void) {
    int n, m, i, k, u, v, c;

    scanf("%d %d", &n, &m);
    for (k = 0; k < m; k++) {
        scanf("%d %d %d", &u, &v, &c);
        for (i = u; i <= v; i++) own[i] = c;
    }

    for (i = 1; i <= n; i++)
        printf(i == n ? "%d\n" : "%d ", own[i]);
    return 0;
}

坑：输出格式是空格分隔、行末不要多空格。上面的写法正好处理了。

延伸：如果 n, m 变成 10⁵（就是 F 题），暴力就会 TLE，那时候要换"倒序 + 跳过已染色"的思路。

F群雄逐鹿（困难版）0 人通过 · 离线逆序 + 并查集 · 你没提交

题意：把 E 题的城市排布从"一条直线"换成"一棵树"（n, m ≤ 10⁵）。每次操作占领 u 到 v 的路径上的所有点，后覆盖先。输出最终归属。

思路（了解即可，不必实现）：

  1. 倒着做：从最后一次操作往回处理。某个点一旦被后面的操作染过，前面就再也改不动它了——所以每个点只会被写一次。
  2. 于是问题变成"每次操作，把路径上还没被染色的点染掉"。
  3. 用并查集维护"跳过已染色点"：find(x) 返回 x 及其祖先中最近的一个未染色点。染完一个点就把它并到它的父亲上，下次直接跳过。
  4. 路径 u→v 拆成 u→lca 和 v→lca 两段，每段用 find 一步步往上跳，总跳次数 ≤ n。

复杂度：O((n + m) α(n) + m log n)（log 来自 LCA）。

诚实说明 这题需要 LCA + 并查集两个东西组合，0 人通过。现在补它性价比极低——知道"倒序处理 + 跳过已染色"这个套路就够了，下次见到类似的"后覆盖先、问最终状态"能有印象。

G校园寻路9 人通过 · BFS 邻接表 · 你 AC 了（130 分钟，1 WA）

题意：n 个点 m 条边的无向图（n, m ≤ 2×10⁵），每条边长度都是 1。求 s 到 t 的最少边数，不可达输出 -1。图可能有重边、可能不连通。

思路：边权全是 1 → BFS。但 2×10⁵ 个点不能开邻接矩阵（4×10¹⁰ 个格子），必须用邻接表：head[] + to[] + nxt[] 三个数组。

G.c
#include <stdio.h>

#define MAXN 200005
#define MAXE 400005

int head[MAXN], to[MAXE], nxt[MAXE], ecnt = 0;
int q[MAXN], dist[MAXN];

void add(int u, int v) {
    to[++ecnt] = v; nxt[ecnt] = head[u]; head[u] = ecnt;
}

int main(void) {
    int n, m, i, s, t, u, v, h, tail = 0;

    scanf("%d %d", &n, &m);
    for (i = 0; i < m; i++) {
        scanf("%d %d", &u, &v);
        add(u, v); add(v, u);        /* 无向图，加两条 */
    }
    scanf("%d %d", &s, &t);

    for (i = 1; i <= n; i++) dist[i] = -1;
    dist[s] = 0; q[tail++] = s;

    for (h = 0; h < tail; h++) {
        int x = q[h];
        for (int e = head[x]; e; e = nxt[e]) {
            int y = to[e];
            if (dist[y] != -1) continue;
            dist[y] = dist[x] + 1;
            q[tail++] = y;
        }
    }

    printf("%d\n", dist[t]);
    return 0;
}

坑：① 无向图每条边要加两次，to/nxt 数组开 2m；② dist 初始化为 -1，正好当"没访问过"的标记，最后不可达时输出 -1 天然正确；③ 队列开 n+1 就够（每个点只进队一次）。

H3355033631 人通过 · 枚举 + 条数判定 · 你没提交 ← 最大的遗憾

题意：给区间 [l, r]（r ≤ 10⁹，且 r-l ≤ 10⁸），数里面有多少个"伪完全数"。伪完全数满足下面三条性质中的至少两条：

  • 性质 1：尾数是 6 或 28；
  • 性质 2：除以 9 余 1（6 除外）；
  • 性质 3：是完全数。

思路：10⁹ 以内的完全数只有 5 个：6, 28, 496, 8128, 33550336（题目标题就是这个数）。所以性质 3 可以写死成 5 个比较。剩下就是一个循环 + 三个判断。

H.c
#include <stdio.h>

int perfect(int x) {
    return x == 6 || x == 28 || x == 496 || x == 8128 || x == 33550336;
}

int main(void) {
    long long l, r, i;
    int ans = 0;

    scanf("%lld %lld", &l, &r);

    for (i = l; i <= r; i++) {
        int p1 = (i % 10 == 6) || (i % 100 == 28);
        int p2 = (i % 9 == 1);
        int p3 = perfect((int)i);
        if (p1 + p2 + p3 >= 2) ans++;
    }

    printf("%d\n", ans);
    return 0;
}

样例验证：1 100 → 3（6、28、46）；1 900 → 12（6、28、46、136、226、316、406、496、586、676、766、856）。

坑：① 三条性质是"至少满足两条"，不是"满足任一"——用 p1+p2+p3 >= 2 最省事；② 循环最多 10⁸ 次，必须用 long long 的 i 还是 int 都行，但判断别做除法；③ 如果想更快，可以加一句"尾数既不是 6 也不是 28 就直接跳过"，只在完全数上再检查。

这题本该是你的 31 人通过，全场第 3 简单。不需要任何算法，只需要把三条性质翻译成三个判断。你当时连题都没读。

I魔术师的试炼1 人通过 · 模拟洗牌 · 你没提交

题意：21 张牌摆成 3 行 7 列（给定初始摆法，1~21 各出现一次）。做三轮"魔术洗牌"，每轮观众告诉你目标牌当前在第 R 行（0/1/2）。三轮之后，求出目标牌的编号。

每轮三步：① 按行收成三堆；② 把第 R 行那堆放到三堆的正中间，其余两堆保持相对顺序；③ 重新发牌：第 i 张（从 1 数）放到第 (i-1)%3 行、第 (i-1)/3 列。

思路：别想着"猜是哪张牌"。反过来——每轮都会把候选缩小到一行。所以：先把第 R₀ 行的 7 张全当候选，做一次洗牌，和第 R₁ 行取交集，再洗一次，和第 R₂ 行取交集，最后洗一次，剩下的那张就是答案。

I.c
#include <stdio.h>
#include <string.h>

int g[3][7];
int cand[22];                 /* cand[x]=1 表示号码 x 还是候选 */

void shuffle(int R) {
    int pile[3][7], deck[21], r, c;
    int others[2], oi = 0;

    for (r = 0; r < 3; r++)
        for (c = 0; c < 7; c++) pile[r][c] = g[r][c];

    for (r = 0; r < 3; r++) if (r != R) others[oi++] = r;

    /* 第 R 堆放中间 */
    for (c = 0; c < 7; c++) deck[c]         = pile[others[0]][c];
    for (c = 0; c < 7; c++) deck[7 + c]     = pile[R][c];
    for (c = 0; c < 7; c++) deck[14 + c]    = pile[others[1]][c];

    /* 重新发牌：第 i 张 -> (i%3, i/3) */
    for (r = 0; r < 3; r++)
        for (c = 0; c < 7; c++) g[r][c] = deck[r + 3 * c];
}

int main(void) {
    int i, j, k, R[3], r, c;

    for (i = 0; i < 3; i++)
        for (j = 0; j < 7; j++) scanf("%d", &g[i][j]);
    for (k = 0; k < 3; k++) scanf("%d", &R[k]);

    for (c = 0; c < 7; c++) cand[g[R[0]][c]] = 1;   /* 第一轮前：必在 R[0] 行 */

    for (k = 0; k < 3; k++) {
        shuffle(R[k]);
        if (k < 2) {                                /* 和下一轮的已知行取交集 */
            int keep[21] = {0};
            for (c = 0; c < 7; c++) if (cand[g[R[k+1]][c]]) keep[g[R[k+1]][c]] = 1;
            memcpy(cand, keep, sizeof(keep));
        }
    }

    for (i = 1; i <= 21; i++) if (cand[i]) { printf("%d\n", i); break; }
    return 0;
}

样例验证：初始按行 1..21，三轮 R = 0,1,2 → 输出 7。

坑：① 收牌顺序是"第 R 堆放中间，另外两堆保持它们原本的先后"（剩下两行里小的在上面）；② 发牌是按列填（i%3 行、i/3 列），不是按行填；③ 三轮之后候选必然只剩 1 张，这是这个魔术的原理。

J魔术师的终极试炼0 人通过 · 数学推导 · 你没提交

题意：R 行 C 列（R, C 都是奇数，≤ 10¹⁸）。问最少几轮洗牌，才能无论观众选哪张牌都唯一确定它的位置。样例 3 9 → 3。

思路：每一轮观众只能提供"它在第几行"这一个信息，也就是 R 种可能。k 轮最多区分 R^k 种情况，而牌有 R×C 张。所以必须 R^k ≥ R×C。
从样例看，这个下界就是答案：3³ = 27 ≥ 3×9 = 27，所以 3 轮。

J.c（思路正确性未完全验证，仅供参考）
#include <stdio.h>

int main(void) {
    long long R, C;
    __int128 limit = 1, cur = 1;
    int k = 0;

    scanf("%lld %lld", &R, &C);
    limit = (__int128)R * C;          /* 1e36，必须 __int128 */

    while (cur < limit) { cur *= R; k++; }

    printf("%d\n", k);
    return 0;
}

说明：全场只有 1 次提交、0 人通过，说明"这个下界能不能取到"需要严格的构造证明。当作数学阅读材料即可。

K星门密码0 人通过 · 子集和 SOS DP · 你没提交

题意：给 n 个数（n ≤ 10⁵，aᵢ ≤ 10⁶），数有多少对 i≠j 满足 aᵢ ⊕ aⱼ = aᵢ + aⱼ。

关键一步：a + b = (a ⊕ b) + 2(a & b)，所以 a ⊕ b = a + b 等价于 a & b == 0（二进制没有同一位同时是 1）。

思路：aᵢ ≤ 10⁶ < 2²⁰，所以值域只有 2²⁰ ≈ 100 万。
① 统计每个值出现几次 cnt[v]；
② 用 SOS DP 求 f[mask] = 有多少个数是 mask 的子集；
③ 对每个值 v，能和它配对的数的个数就是 f[全1 ^ v]（v 的补集的子集）；
④ 加起来是"有序对"，去掉自身配对后除以 2。

K.c
#include <stdio.h>

#define B   20
#define SZ  (1 << B)

int cnt[SZ];
int f[SZ];

int main(void) {
    int n, i, b, v;
    long long ans = 0;

    scanf("%d", &n);
    for (i = 0; i < n; i++) { scanf("%d", &v); cnt[v]++; }

    for (v = 0; v < SZ; v++) f[v] = cnt[v];
    for (b = 0; b < B; b++)
        for (v = 0; v < SZ; v++)
            if (v & (1 << b)) f[v] += f[v ^ (1 << b)];

    for (v = 0; v < SZ; v++)
        if (cnt[v]) ans += (long long)cnt[v] * f[(SZ - 1) ^ v];

    ans -= cnt[0];      /* 唯一的"自己配自己"只发生在 0 上 */
    ans /= 2;           /* 有序对 -> 无序对 */

    printf("%lld\n", ans);
    return 0;
}

样例验证：1 2 3 4 8 → 8。

说明：SOS DP（子集和变换）是这题的核心，20×2²⁰ ≈ 2×10⁷ 次操作，1 秒够用。这道题没法暴力，0 人通过可以理解。

L八级钳工3 人通过 · 解方程 · 你没提交 ← 送分题

题意：给三个轴心坐标 A、B、C，找三个齿轮半径，使它们两两外切。无解输出 -1，有解输出一行三个半径（升序，误差 10⁻⁶ 内都算对）。

思路："外切"就是圆心距等于两个半径之和：

d_AB = r₁ + r₂    d_BC = r₂ + r₃    d_CA = r₃ + r₁

三元一次方程组，解出来很整齐：

r₁ = (d_AB + d_CA − d_BC) / 2
r₂ = (d_AB + d_BC − d_CA) / 2
r₃ = (d_BC + d_CA − d_AB) / 2

只要有一个半径 ≤ 0，就无解（三点共线时就是这样）。

L.c
#include <stdio.h>
#include <math.h>

int main(void) {
    double x1, y1, x2, y2, x3, y3, d12, d23, d31, r1, r2, r3, t;

    scanf("%lf %lf %lf %lf %lf %lf",
          &x1, &y1, &x2, &y2, &x3, &y3);

    d12 = hypot(x2 - x1, y2 - y1);
    d23 = hypot(x3 - x2, y3 - y2);
    d31 = hypot(x1 - x3, y1 - y3);

    r1 = (d12 + d31 - d23) / 2;
    r2 = (d12 + d23 - d31) / 2;
    r3 = (d23 + d31 - d12) / 2;

    if (r1 <= 0 || r2 <= 0 || r3 <= 0) { printf("-1\n"); return 0; }

    if (r1 > r2) { t = r1; r1 = r2; r2 = t; }
    if (r2 > r3) { t = r2; r2 = r3; r3 = t; }
    if (r1 > r2) { t = r1; r1 = r2; r2 = t; }

    printf("%.6f %.6f %.6f\n", r1, r2, r3);
    return 0;
}

样例验证：0 0 1 0 1 1 → 0.292893 0.707107 0.707107；0 0 1 0 2 0 → -1。

坑：坐标到 10⁸，距离用 double 和 hypot 就够（hypot 自带溢出保护）；判"无解"用 r <= 0，别用 == 0 去比浮点。

这题真的只是解方程 只有 3 人通过，但代码不到 20 行，没有几何库、没有精度陷阱。大多数人被"齿轮 / 几何"的描述吓退了。

M电影院热度统计18 人通过 · 差分 + 前缀和 · 你 AC 了（167 分钟，5 WA）

题意：n 个时间段各有一个初始热度（n ≤ 10⁵）。T 次操作（T ≤ 10⁵），每次给区间 [P,Q] 全体加 R（R 可以是负数，|R| ≤ 10⁶）。最后问区间 [l,r] 的热度总和。

思路：这就是你练过的差分 + 前缀和，一步都没变：

  1. 每次区间加：d[P] += R; d[Q+1] -= R;（O(1) 一次操作）
  2. 全部操作完，对 d 求前缀和并累加到原数组上，得到最终热度；
  3. 再对最终热度求前缀和，回答查询：S[r] - S[l-1]。
M.c
#include <stdio.h>

#define MAXN 100005

long long a[MAXN];     /* 最终热度 */
long long d[MAXN];     /* 差分数组，注意开 n+2 的余量 */
long long S[MAXN];     /* 最终热度的前缀和 */

int main(void) {
    int n, T, i, l, r;
    long long R, P, Q, cur = 0;

    scanf("%d", &n);
    for (i = 1; i <= n; i++) scanf("%lld", &a[i]);

    scanf("%d", &T);
    for (i = 0; i < T; i++) {
        scanf("%lld %lld %lld", &R, &P, &Q);
        if (P > n) continue;
        if (Q > n) Q = n;              /* 题目给的上界比 n 大，钳一下 */
        d[P] += R;
        d[Q + 1] -= R;
    }

    for (i = 1; i <= n; i++) { cur += d[i]; a[i] += cur; }
    for (i = 1; i <= n; i++) S[i] = S[i-1] + a[i];

    scanf("%d %d", &l, &r);
    printf("%lld\n", S[r] - S[l-1]);
    return 0;
}

样例验证：初始 20 -10 30 -15 40 60，四次操作后是 5 0 80 35 70 60，查询 [3,6] → 245。

坑：① 全部用 long long——10⁵ 次 × 10⁶ 能到 10¹¹，int 必炸；② 题目里 Q 最大 10⁶ 而 n 只有 10⁵，不钳一下会越界（这可能是你 5 次 WA 的原因）；③ 差分数组要开 n+2，因为会写到 d[Q+1]。

N旅行中的游戏1 人通过 · 博弈 + 分解质因数 · 你没提交

题意：t 组询问（t ≤ 10⁵，n ≤ 10⁹）。n 是巨石大小。轮到谁时，若 n 是质数或 1，他可以直接砸碎它并获胜；否则他必须选一个 1 < x < n 且 x | n，把 n 变成 n/x。问先手（小怪兽）能否取胜。

思路：只看 n 的质因数个数（带重复），记作 Ω(n)：

  • Ω(n) = 0（n=1）或 1（质数）→ 当前玩家立刻赢；
  • Ω(n) = 2（n = p·q）→ 无论怎么选，结果都是质数，对手立刻赢，所以当前玩家必败；
  • Ω(n) ≥ 3 → 可以一次消掉 Ω(n)−2 个质因子，把局面变成 Ω=2 交给对手 → 当前玩家必胜。

结论：Ω(n) == 2 则先手必败，否则先手必胜。

N.c
#include <stdio.h>

int omega(int n) {
    int c = 0, p;
    for (p = 2; (long long)p * p <= n; p++) {
        while (n % p == 0) { n /= p; c++; }
        if (c >= 3) return 3;      /* 只要知道 >=3，可以早退 */
    }
    if (n > 1) c++;
    return c;
}

int main(void) {
    int t, n;
    scanf("%d", &t);
    while (t--) {
        scanf("%d", &n);
        printf(omega(n) == 2 ? "Sakurazuihaole\n" : "Sakurazhenlihai\n");
    }
    return 0;
}

样例验证：24 = 2³·3 → Ω=4 → Sakurazhenlihai；14 = 2·7 → Ω=2 → Sakurazuihaole。

坑：① 试除到 p*p > n 就停，最后剩下的 n > 1 本身是一个质因子；② 计数到 3 就可以提前返回，否则 t 组 × 试除会慢；③ 输出字符串一个字都不能差，两个串长得很像。
